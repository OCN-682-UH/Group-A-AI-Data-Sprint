# ---------------------------------------------------------------------------
# Collapses ONLY cosmetic spelling differences - ʻokina, smart quotes,
# backtick, macrons, case, spacing. Used for matching and deduplication.
# Never write its output to the master sheet: the correct Hawaiian spelling
# comes from site_lookup.csv.
# ---------------------------------------------------------------------------
match_key <- function(x) {
  x |>
    str_replace_all("[\u02BB\u2018\u2019`']", "") |>
    stri_trans_general("Latin-ASCII") |>
    str_to_lower() |>
    str_squish()
}

# ---------------------------------------------------------------------------
# STEP 1 - read. Header row is detected, not assumed, because the title row
# above it isn't consistent between the two files.
# ---------------------------------------------------------------------------
read_2024 <- function(path) {
  probe  <- read_csv(path, col_names = FALSE, n_max = 4, show_col_types = FALSE,
                     col_types = cols(.default = col_character()))
  skip_n <- which(apply(probe, 1, \(r) any(trimws(r) == "Date", na.rm = TRUE)))[1] - 1

  read_csv(path, skip = skip_n, show_col_types = FALSE,
           col_types = cols(.default = col_character())) |>
    clean_names() |>
    mutate(source_file = basename(path))
}

# STEP 2 - harmonise column names. The copy abbreviates two species that main
# spells out; without this, bind_rows() makes four columns for two organisms.
# Both files also carry a nameless trailing column (x96 / x91).
# ---------------------------------------------------------------------------
tidy_2024 <- function(df) {
  df |>
    select(-matches("^x\\d+$")) |>
    rename(
      haminoea_cymbalum_green_snail = any_of("haminoea_cymbalum"),
      snail_dendropoma_gregarium    = any_of("snail_dendropoma")
    )
}

clean_2024 <- dedup_2024 |>
  mutate(
    note     = if_else(str_detect(count_raw, "^\\d+$"), NA_character_, count_raw),
    count    = suppressWarnings(as.numeric(count_raw)),
    date     = mdy(date),
    transect = suppressWarnings(as.integer(number_t)),
    position = suppressWarnings(as.numeric(position_number_on_transect_m)),
    year     = year(date),
    format   = "B"
  ) |>
  select(format, year, date, site_raw = location, site_key,
         coder_id = data_entry_by, transect, position,
         taxon_raw, count, note, source_file)
